第二步：判斷一個數是不是質數
step2.py
def is_prime(n):
    if n < 2:
        return False
    for i in range(2, n):
        if n % i == 0:   # 有一個數能整除它
            return False  # 就不是質數
    return True

print(is_prime(7))   # True
print(is_prime(10))  # False
第三步：印出 100 以內所有質數

for n in range(2, 101):
    if is_prime(n):
        primes.append(n)

print(primes)
print(f'共 {len(primes)} 個')

程式碼
rgb_palette.py
import matplotlib.pyplot as plt
import matplotlib.patches as patches

r = int(input('Red (0–255)：'))
g = int(input('Green (0–255)：'))
b = int(input('Blue (0–255)：'))

if r > b + 30:
    feel = '偏暖色（紅橙黃方向）'
elif b > r + 30:
    feel = '偏冷色（藍紫方向）'
elif abs(r-g) < 20 and abs(g-b) < 20:
    feel = '接近灰色（三色相近）'
else:
    feel = '中性色'

print(f'你的顏色：RGB({r}, {g}, {b})  →  {feel}')

fig, ax = plt.subplots(figsize=(3, 3))
color = (r/255, g/255, b/255)
rect = patches.Rectangle((0,0), 1, 1, color=color)
ax.add_patch(rect)
ax.set_xlim(0,1); ax.set_ylim(0,1); ax.axis('off')
ax.set_title(f'RGB({r}, {g}, {b})')
plt.show()

程式碼
equation_solver.py
a = float(input('輸入 a（x 的係數）：'))
b = float(input('輸入 b（常數項）：'))

print(f'你的方程式是：{a}x + {b} = 0')

if a == 0:
    if b == 0:
        print('任何 x 都是解（無限多解）')
    else:
        print('無解')
else:
    x = -b / a
    print(f'解：x = {x}')
    # 驗算
    check = a * x + b
    print(f'驗算：{a} × {x} + {b} = {check:.6f}')

這次目標

輸入不等式 ax+b>c，程式解出解集合並用數線圖顯示出來

📋 給家長說（備課重點）
  • 先手算：2x + 1 > 5，x > 2（移項、除以正數不等號方向不變）
  • 問他：「這個解怎麼在數線上畫出來？」（從 2 往右的射線）
  • 特別注意：除以負數時不等號要反向，這是很多人搞錯的地方
程式碼
inequality.py
import matplotlib.pyplot as plt

a = float(input('ax + b > c，輸入 a：'))
b = float(input('輸入 b：'))
c = float(input('輸入 c：'))

rhs = c - b

if a == 0:
    print('無法解（a 不能是 0）')
elif a > 0:
    boundary = rhs / a
    direction = 'right'
    print(f'解：x > {boundary:.2f}')
else:
    boundary = rhs / a
    direction = 'left'
    print(f'解：x < {boundary:.2f}  ← 除以負數，方向反轉！')

fig, ax = plt.subplots(figsize=(8, 2))
ax.axhline(0, color='black', linewidth=1.5)
ax.set_xlim(boundary-5, boundary+5)
ax.plot(boundary, 0, 'o', markersize=12, color='white',
       markeredgecolor='blue', markeredgewidth=2)  # 空心圓=不包含
ax.set_yticks([])
ax.set_title(f'{a}x + {b} > {c}　解集合')
plt.tight_layout(); plt.show()
X 來做做看

  • 測試：-3x + 6 > 0（注意除以負數！答案：x < 2）
  • 改成 ≥（大於等於），圓點改成實心
口頭確認

速度比較圖

一次函數 = 等速運動
⏱ 時間

🛠 工具
Colab + matplotlib
📐 數學
一次函數 y=kx+b，斜率（七下）
🔬 物理
等速直線運動 d=vt
🎯
這次目標

畫出不同速度的 d-t 圖，用眼睛看出「斜率就是速度」

📋 給家長說（備課重點）
  • 這堂課是數學和物理的橋樑：d = v × t 和 y = k × x 是同一件事
  • 先口頭問：「一輛車時速 60 km，開 2 小時走多遠？」（120 km）
  • d=vt 中的 v 就是斜率：速度越快，d-t 圖的斜率越大，線越陡
  • 讓他看圖然後說：「哪輛車最快？你怎麼看出來的？」
程式碼
speed_comparison.py
import matplotlib.pyplot as plt
import numpy as np

t = np.linspace(0, 10, 100)

v1, v2, v3 = 10, 25, 50  # 速度 m/s

plt.figure(figsize=(7, 5))
plt.plot(t, v1*t, label=f'v = {v1} m/s（慢）', color='green')

程式碼
projectile_v2.py
import matplotlib.pyplot as plt
import numpy as np
from ipywidgets import interact, FloatSlider, IntSlider

def projectile(v0=20, theta=45):
    g     = 9.8
    angle = np.radians(theta)
    T     = 2 * v0 * np.sin(angle) / g
    t     = np.linspace(0, T, 300)
    x     = v0 * np.cos(angle) * t
    y     = v0 * np.sin(angle) * t - 0.5 * g * t**2

    t_top = v0 * np.sin(angle) / g
    x_top = v0 * np.cos(angle) * t_top
    y_top = v0 * np.sin(angle) * t_top - 0.5 * g * t_top**2

    plt.figure(figsize=(8, 4))
    plt.plot(x, y, 'b-', linewidth=2)
    plt.plot(x_top, y_top, 'r^', markersize=12,
             label=f'最高點 ({x_top:.1f}, {y_top:.1f})')
    plt.plot(x[-1], 0, 'gs', markersize=12,
             label=f'落地點 x={x[-1]:.1f} m')
    plt.legend(); plt.grid(True, linestyle='--', alpha=0.5)
    plt.axhline(0, color='gray')
    plt.title(f'拋體模擬 v₀={v0} m/s, θ={theta}°')
    plt.show()
    print(f'最大距離：{x[-1]:.2f} m  |  最大高度：{y_top:.2f} m')

interact(projectile,
         v0=FloatSlider(min=5, max=50, step=1, value=20),
         theta=IntSlider(min=10, max=80, step=1, value=45))

import matplotlib.pyplot as plt
import numpy as np

def roll_dice(n):
    rolls = np.random.randint(1, 7, size=n)
    return [(rolls == i).sum() / n for i in range(1, 7)]

trials = [10, 100, 1000, 10000]
fig, axes = plt.subplots(1, 4, figsize=(14, 4))

for ax, n in zip(axes, trials):
    counts = roll_dice(n)
    ax.bar(range(1, 7), counts, color='steelblue', alpha=0.7)
    ax.axhline(1/6, color='red', linestyle='--', label='理論值 1/6')
    ax.set_ylim(0, 0.5)
    ax.set_title(f'n = {n}')
    ax.legend(fontsize=8)

plt.suptitle('擲骰子次數與公平性', fontsize=13)
plt.tight_layout()
plt.show()

📋 給家長說（備課重點）

  • 估算初速度：v₀ = √(xg / sin(2θ))

程式碼
soccer_physics.py
import matplotlib.pyplot as plt
import numpy as np

x_measured = float(input('你踢出去的距離（公尺）：'))
theta      = float(input('估計的仰角（度）：'))
g          = 9.8
angle      = np.radians(theta)

# 反推初速度：x = v₀² × sin(2θ) / g
v0 = np.sqrt(x_measured * g / np.sin(2 * angle))
print(f'你的踢球初速度：{v0:.1f} m/s（{v0*3.6:.1f} km/h）')

T     = 2 * v0 * np.sin(angle) / g
t     = np.linspace(0, T, 300)
x_sim = v0 * np.cos(angle) * t
y_sim = v0 * np.sin(angle) * t - 0.5 * g * t**2

plt.figure(figsize=(8, 4))
plt.plot(x_sim, y_sim, 'b-', linewidth=2, label='模擬軌跡')
plt.axvline(x_measured, color='red', linestyle='--',
            label=f'實際落點 {x_measured} m')
plt.xlabel('距離（公尺）'); plt.ylabel('高度（公尺）')
plt.title(f'我的足球軌跡（v₀≈{v0:.1f} m/s, θ={theta}°）')
plt.legend(); plt.grid(True, linestyle='--', alpha=0.5)
plt.axhline(0, color='gray')
plt.show()
